  ```typescript
  // src/components/instagram/feed-widget.tsx
  // Instagram feed widget using Behold
  // Client-side only (SSR disabled to prevent hydration errors)
  // RELEVANT FILES: src/app/[locale]/results/page-client.tsx

  "use client";

  import dynamic from "next/dynamic";
  import { motion } from "framer-motion";
  import { Instagram } from "lucide-react";

  // Dynamic import with SSR disabled - CRITICAL for Behold widget
  // Behold uses browser-only APIs that cause hydration errors if SSR'd
  const BeholdWidget = dynamic(
    () => import("@behold/react").then((mod) => mod.default || mod),
    {
      ssr: false,
      loading: () => (
        <div className="min-h-[400px] flex items-center justify-center">
          <div className="text-center">
            <Instagram className="w-12 h-12 text-primary/50 mx-auto mb-4 animate-pulse" />
            <p className="text-sm text-muted-foreground">Loading Instagram feed...</p>
          </div>
        </div>
      ),
    }
  );

  interface InstagramFeedProps {
    feedId?: string;
    sectionTitle?: string;
    sectionSubtitle?: string;
    instagramHandle?: string;
  }

  export function InstagramFeed({
    feedId = process.env.NEXT_PUBLIC_BEHOLD_FEED_ID,
    sectionTitle = "Follow Us",
    sectionSubtitle = "See what we're up to on Instagram",
    instagramHandle = "@roseyco.official",
  }: InstagramFeedProps) {
    // If no feed ID configured, show placeholder
    if (!feedId) {
      return (
        <section className="py-24 md:py-32 border-t border-border/30">
          <div className="container">
            <motion.div
              className="text-center"
              initial={{ opacity: 0, y: 40 }}
              whileInView={{ opacity: 1, y: 0 }}
              viewport={{ once: true, margin: "-100px" }}
              transition={{ duration: 0.7 }}
            >
              <h2 className="text-4xl md:text-5xl lg:text-6xl tracking-tight mb-6">
                {sectionTitle}{" "}
                <span className="gradient-text">{instagramHandle}</span>
              </h2>
              <p className="text-xl md:text-2xl text-muted-foreground max-w-3xl mx-auto mb-8">
                {sectionSubtitle}
              </p>
              <a
                href={`https://instagram.com/${instagramHandle.replace("@", "")}`}
                target="_blank"
                rel="noopener noreferrer"
                className="inline-flex items-center gap-2 px-6 py-3 rounded-full bg-primary/10 text-primary border border-primary/20 hover:bg-primary/20 transition-colors"
              >
                <Instagram className="w-5 h-5" />
                Follow on Instagram
              </a>
            </motion.div>
          </div>
        </section>
      );
    }

    return (
      <section className="py-24 md:py-32 border-t border-border/30">
        <div className="container">
          <motion.div
            className="text-center mb-12"
            initial={{ opacity: 0, y: 40 }}
            whileInView={{ opacity: 1, y: 0 }}
            viewport={{ once: true, margin: "-100px" }}
            transition={{ duration: 0.7 }}
          >
            <h2 className="text-4xl md:text-5xl lg:text-6xl tracking-tight mb-6">
              {sectionTitle}{" "}
              <span className="gradient-text">{instagramHandle}</span>
            </h2>
            <p className="text-xl md:text-2xl text-muted-foreground max-w-3xl mx-auto">
              {sectionSubtitle}
            </p>
          </motion.div>

          <motion.div
            className="min-h-[500px]"
            initial={{ opacity: 0, y: 20 }}
            whileInView={{ opacity: 1, y: 0 }}
            viewport={{ once: true }}
            transition={{ delay: 0.3, duration: 0.6 }}
          >
            <BeholdWidget feedId={feedId} />
          </motion.div>

          <motion.div
            className="text-center mt-8"
            initial={{ opacity: 0 }}
            whileInView={{ opacity: 1 }}
            viewport={{ once: true }}
            transition={{ delay: 0.5, duration: 0.5 }}
          >
            <a
              href={`https://instagram.com/${instagramHandle.replace("@", "")}`}
              target="_blank"
              rel="noopener noreferrer"
              className="inline-flex items-center gap-2 text-sm text-muted-foreground hover:text-primary transition-colors"
            >
              <Instagram className="w-4 h-4" />
              View more on Instagram
            </a>
          </motion.div>
        </div>
      </section>
    );
  }
  ```

  Key features:
  - Dynamic import with ssr: false prevents hydration errors
  - Loading state shows Instagram icon with pulse animation
  - Graceful fallback if no feedId configured (shows link to Instagram)
  - Framer Motion animations consistent with Results page
  - Links to Instagram profile with proper noopener/noreferrer
  </action>

  - TypeScript compiles without errors
  - Component exports InstagramFeed
  - Uses dynamic import with ssr: false
