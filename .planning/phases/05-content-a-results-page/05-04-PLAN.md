  ```typescript
  // src/app/api/guide-download/route.ts
  // API route for free guide download form
  // Saves lead to Supabase, sends guide via email with public download URL
  // Pattern: Supabase Storage URL (not attachment) for deliverability
  // RELEVANT FILES: src/app/api/contact/route.ts (similar pattern)

  import { NextRequest, NextResponse } from 'next/server';
  import { createServerClient } from '@/lib/supabase';
  import { resend, FROM_EMAIL } from '@/lib/resend';

  interface GuideDownloadData {
    email: string;
    fullName: string;
    company?: string;
    company_website?: string; // Honeypot field
  }

  // Guide configuration
  const GUIDE_CONFIG = {
    name: 'Why Meta Ads Are The Greatest',
    fileName: 'why-meta-ads-are-the-greatest.pdf',
    bucket: 'guides',
  };

  export async function POST(request: NextRequest) {
    try {
      const data: GuideDownloadData = await request.json();

      // Honeypot check - if filled, it's a bot
      if (data.company_website) {
        return NextResponse.json(
          { error: 'Invalid submission' },
          { status: 400 }
        );
      }

      // Validate required fields
      if (!data.fullName || !data.email) {
        return NextResponse.json(
          { error: 'Name and email are required' },
          { status: 400 }
        );
      }

      // Validate email format
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(data.email)) {
        return NextResponse.json(
          { error: 'Invalid email address' },
          { status: 400 }
        );
      }

      const supabase = createServerClient();

      // 1. Save lead to database
      const { error: dbError } = await supabase
        .from('free_guide_downloads')
        .insert({
          email: data.email,
          full_name: data.fullName,
          company: data.company || null,
          guide_name: GUIDE_CONFIG.name,
          created_at: new Date().toISOString(),
        });

      if (dbError) {
        console.error('Supabase error:', dbError);
        // Continue even if DB fails - still try to send email
      }

      // 2. Get public URL for guide PDF from Supabase Storage
      const { data: storageData } = supabase.storage
        .from(GUIDE_CONFIG.bucket)
        .getPublicUrl(GUIDE_CONFIG.fileName);

      const downloadUrl = storageData.publicUrl;

      // 3. Send email with download link (NOT attachment)
      if (resend) {
        try {
          await resend.emails.send({
            from: FROM_EMAIL,
            to: data.email,
            subject: `Your Free Guide: ${GUIDE_CONFIG.name}`,
            html: `
              <!DOCTYPE html>
              <html>
              <head>
                <meta charset="utf-8">
                <meta name="viewport" content="width=device-width, initial-scale=1.0">
              </head>
              <body style="font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif; line-height: 1.6; color: #333; max-width: 600px; margin: 0 auto; padding: 20px;">
                <div style="text-align: center; margin-bottom: 30px;">
                  <h1 style="color: #DC3C46; margin-bottom: 10px;">Your Free Guide is Ready!</h1>
                </div>

                <p>Hi ${data.fullName.split(' ')[0]},</p>

                <p>Thanks for downloading <strong>"${GUIDE_CONFIG.name}"</strong>!</p>

                <p>Click the button below to download your guide:</p>

                <div style="text-align: center; margin: 30px 0;">
                  <a href="${downloadUrl}" style="display: inline-block; background: linear-gradient(135deg, #DC3C46 0%, #8B1538 100%); color: white; text-decoration: none; padding: 15px 30px; border-radius: 8px; font-weight: 600; font-size: 16px;">Download Your Guide</a>
                </div>

                <p style="font-size: 14px; color: #666;">
                  If the button doesn't work, copy and paste this link into your browser:<br>
                  <a href="${downloadUrl}" style="color: #DC3C46; word-break: break-all;">${downloadUrl}</a>
                </p>

                <hr style="border: none; border-top: 1px solid #eee; margin: 30px 0;">

                <div style="background: linear-gradient(135deg, rgba(220,60,70,0.1) 0%, rgba(139,21,56,0.1) 100%); padding: 20px; border-radius: 8px; text-align: center;">
                  <h3 style="margin-top: 0; color: #DC3C46;">Want Us to Handle This For You?</h3>
                  <p style="margin-bottom: 15px; font-size: 14px;">Ready to implement these strategies? Let's chat about how we can grow your business.</p>
                  <a href="https://roseyco.com/contact" style="display: inline-block; background: #DC3C46; color: white; text-decoration: none; padding: 12px 24px; border-radius: 6px; font-weight: 600;">Book a Free Strategy Call</a>
                </div>

                <p style="margin-top: 30px;">Best regards,<br><strong>The Rosey Co Team</strong></p>

                <hr style="border: none; border-top: 1px solid #eee; margin: 30px 0;">

                <p style="font-size: 12px; color: #999; text-align: center;">
                  You received this email because you downloaded a guide from Rosey Co.<br>
                  If you didn't request this, please ignore this email.
                </p>
              </body>
              </html>
            `,
          });
        } catch (emailError) {
          console.error('Resend error:', emailError);
          // Return success anyway - lead is saved
        }
      }

      return NextResponse.json({ success: true });
    } catch (error) {
      console.error('Guide download error:', error);
      return NextResponse.json(
        { error: 'Failed to process request' },
        { status: 500 }
      );
    }
  }
  ```

  ```typescript
  // Free guide download page - Client component with form
  // RELEVANT FILES: migration-content/pages/FreeGuide.tsx (migrated from)

  "use client";

  import { useState } from "react";
  import Link from "next/link";
  import { motion } from "framer-motion";
  import { CheckCircle, Download, Star, ArrowRight } from "lucide-react";
  import { Button } from "@/components/ui/button";
  import { Input } from "@/components/ui/input";
  import { Label } from "@/components/ui/label";
  import { isValidLocale, LocaleCode } from "@/lib/locales";

  interface PageProps {
    params: { locale: string };
  }

  const benefits = [
    "Discover why Meta Ads outperform all other advertising platforms",
    "Learn the secret targeting strategies used by top marketers",
    "Get access to proven ad templates that convert",
    "Understand the psychology behind high-converting Meta Ads",
    "Master the art of audience research and segmentation",
  ];

  export default function FreeGuidePageClient({ params }: PageProps) {
    const { locale } = params;
    const validLocale: LocaleCode = isValidLocale(locale) ? locale : "us";

    const [formData, setFormData] = useState({
      fullName: "",
      email: "",
      company: "",
      company_website: "", // Honeypot
    });
    const [isSubmitting, setIsSubmitting] = useState(false);
    const [isSubmitted, setIsSubmitted] = useState(false);
    const [error, setError] = useState<string | null>(null);

    const handleSubmit = async (e: React.FormEvent) => {
      e.preventDefault();
      setIsSubmitting(true);
      setError(null);

      try {
        const response = await fetch("/api/guide-download", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(formData),
        });

        const data = await response.json();

        if (!response.ok) {
          throw new Error(data.error || "Failed to submit form");
        }

        setIsSubmitted(true);
      } catch (err: any) {
        setError(err.message || "Something went wrong. Please try again.");
      } finally {
        setIsSubmitting(false);
      }
    };

    // Success state
    if (isSubmitted) {
      return (
        <div className="min-h-screen">
          <div className="container py-16 md:py-24">
            <motion.div
              className="max-w-2xl mx-auto text-center"
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.6 }}
            >
              <CheckCircle className="w-20 h-20 text-brand-green mx-auto mb-6" />
              <h1 className="text-4xl md:text-5xl font-bold mb-6">Thank You!</h1>
              <p className="text-xl text-muted-foreground mb-8">
                We've sent you an email with your free guide: "Why Meta Ads Are The Greatest"
              </p>
              <p className="text-muted-foreground mb-12">
                Check your inbox (and spam folder) for the download link. If you don't see it within a few minutes, please contact us.
              </p>

              <div className="service-card p-8 mb-8">
                <h2 className="text-2xl font-bold mb-4">Want This Done For You?</h2>
                <p className="text-muted-foreground mb-6">
                  Ready to implement these strategies? Let's chat about how we can grow your business.
                </p>
                <Button asChild size="lg" className="btn-hero">
                  <Link href={`/${validLocale}/contact`} className="flex items-center gap-2">
                    Book a Free Strategy Call <ArrowRight className="w-5 h-5" />
                  </Link>
                </Button>
              </div>

              <Button asChild variant="outline" size="lg">
                <Link href={`/${validLocale}`}>Back to Homepage</Link>
              </Button>
            </motion.div>
          </div>
        </div>
      );
    }

    // Form state
    return (
      <div className="min-h-screen">
        <div className="container py-16 md:py-24">
          <div className="max-w-5xl mx-auto">
            {/* Hero */}
            <motion.div
              className="text-center mb-12"
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.6 }}
            >
              <div className="flex justify-center mb-6">
                {[...Array(5)].map((_, i) => (
                  <Star key={i} className="w-6 h-6 fill-yellow-400 text-yellow-400" />
                ))}
              </div>
              <span className="inline-block px-5 py-2 mb-6 text-sm font-medium tracking-wider uppercase rounded-full bg-primary/10 text-primary border border-primary/20">
                Free Download
              </span>
              <h1 className="text-4xl md:text-5xl lg:text-6xl font-bold mb-6">
                Why Meta Ads Are{" "}
                <span className="gradient-accent-text">The Greatest</span>
              </h1>
              <p className="text-xl text-muted-foreground max-w-2xl mx-auto">
                Discover the data-driven strategies that top businesses use to generate massive ROI with Meta Ads.
              </p>
            </motion.div>

            <div className="grid lg:grid-cols-2 gap-12 items-start">
              {/* Benefits */}
              <motion.div
                initial={{ opacity: 0, x: -20 }}
                animate={{ opacity: 1, x: 0 }}
                transition={{ duration: 0.6, delay: 0.2 }}
              >
                <h2 className="text-2xl font-bold mb-6">What You'll Learn:</h2>
                <div className="service-card p-6">
                  <ul className="space-y-4">
                    {benefits.map((benefit, index) => (
                      <li key={index} className="flex items-start gap-3">
                        <span className="w-2 h-2 rounded-full bg-primary flex-shrink-0 mt-2" />
                        <span className="text-muted-foreground">{benefit}</span>
                      </li>
                    ))}
                  </ul>
                </div>

                <div className="mt-8 p-6 bg-primary/10 rounded-xl border border-primary/20">
                  <div className="flex items-center gap-2 mb-2">
                    <Download className="w-5 h-5 text-primary" />
                    <span className="font-semibold">Instant Download</span>
                  </div>
                  <p className="text-sm text-muted-foreground">
                    Get immediate access to your guide delivered straight to your inbox.
                  </p>
                </div>
              </motion.div>

              {/* Form */}
              <motion.div
                initial={{ opacity: 0, x: 20 }}
                animate={{ opacity: 1, x: 0 }}
                transition={{ duration: 0.6, delay: 0.3 }}
              >
                <div className="service-card p-8">
                  <h3 className="text-2xl font-bold mb-2">Get Your Free Guide</h3>
                  <p className="text-muted-foreground mb-6">
                    Enter your details below to download the guide instantly
                  </p>

                  <form onSubmit={handleSubmit} className="space-y-5">
                    <div>
                      <Label htmlFor="fullName">Full Name *</Label>
                      <Input
                        id="fullName"
                        type="text"
                        value={formData.fullName}
                        onChange={(e) => setFormData({ ...formData, fullName: e.target.value })}
                        required
                        placeholder="Enter your full name"
                        className="mt-1"
                      />
                    </div>

                    <div>
                      <Label htmlFor="email">Email Address *</Label>
                      <Input
                        id="email"
                        type="email"
                        value={formData.email}
                        onChange={(e) => setFormData({ ...formData, email: e.target.value })}
                        required
                        placeholder="Enter your email address"
                        className="mt-1"
                      />
                    </div>

                    <div>
                      <Label htmlFor="company">Company (Optional)</Label>
                      <Input
                        id="company"
                        type="text"
                        value={formData.company}
                        onChange={(e) => setFormData({ ...formData, company: e.target.value })}
                        placeholder="Enter your company name"
                        className="mt-1"
                      />
                    </div>

                    {/* Honeypot field - hidden from users */}
                    <div className="hidden" aria-hidden="true">
                      <Label htmlFor="company_website">Website</Label>
                      <Input
                        id="company_website"
                        type="text"
                        value={formData.company_website}
                        onChange={(e) => setFormData({ ...formData, company_website: e.target.value })}
                        tabIndex={-1}
                        autoComplete="off"
                      />
                    </div>

                    {error && (
                      <p className="text-sm text-red-500">{error}</p>
                    )}

                    <Button
                      type="submit"
                      size="lg"
                      className="w-full btn-hero"
                      disabled={isSubmitting}
                    >
                      {isSubmitting ? "Sending..." : "Download Free Guide"}
                    </Button>

                    <p className="text-xs text-muted-foreground text-center">
                      We respect your privacy. Unsubscribe at any time.
                    </p>
                  </form>
                </div>
              </motion.div>
            </div>
          </div>
        </div>
      </div>
    );
  }
  ```
