  ```typescript
  // src/components/video/results-video-grid.tsx
  // Video grid for Results page portfolio section
  // Reuses HeroVideoPlayer component for consistent behavior
  // RELEVANT FILES: src/components/video/hero-video-player.tsx

  "use client";

  import { motion } from "framer-motion";
  import { HeroVideoPlayer } from "./hero-video-player";
  import { VideoItem } from "@/lib/video-data";
  import { Play } from "lucide-react";

  interface ResultsVideoGridProps {
    videos: VideoItem[];
    sectionTitle?: string;
    sectionSubtitle?: string;
  }

  const staggerContainer = {
    hidden: { opacity: 0 },
    visible: {
      opacity: 1,
      transition: { staggerChildren: 0.15, delayChildren: 0.1 },
    },
  };

  const staggerItem = {
    hidden: { opacity: 0, y: 30 },
    visible: { opacity: 1, y: 0 },
  };

  export function ResultsVideoGrid({
    videos,
    sectionTitle,
    sectionSubtitle,
  }: ResultsVideoGridProps) {
    // Filter out videos without URLs (graceful degradation)
    const availableVideos = videos.filter((v) => v.hlsUrl || v.thumbnailUrl);

    if (availableVideos.length === 0) {
      return null; // Don't render section if no videos available
    }

    return (
      <section className="py-24 md:py-32">
        <div className="container">
          {(sectionTitle || sectionSubtitle) && (
            <motion.div
              className="text-center mb-16 md:mb-20"
              initial={{ opacity: 0, y: 40 }}
              whileInView={{ opacity: 1, y: 0 }}
              viewport={{ once: true, margin: "-100px" }}
              transition={{ duration: 0.7 }}
            >
              {sectionTitle && (
                <h2 className="text-4xl md:text-5xl lg:text-6xl tracking-tight mb-6">
                  {sectionTitle}
                </h2>
              )}
              {sectionSubtitle && (
                <p className="text-xl md:text-2xl text-muted-foreground max-w-3xl mx-auto">
                  {sectionSubtitle}
                </p>
              )}
            </motion.div>
          )}

          <motion.div
            className="grid md:grid-cols-2 lg:grid-cols-3 gap-6 md:gap-8"
            variants={staggerContainer}
            initial="hidden"
            whileInView="visible"
            viewport={{ once: true, margin: "-50px" }}
          >
            {availableVideos.map((video) => (
              <motion.div
                key={video.id}
                variants={staggerItem}
                transition={{ duration: 0.5 }}
                className="group"
              >
                <div className="rounded-2xl p-px bg-gradient-to-br from-border/50 via-border/20 to-border/50 group-hover:from-primary/20 group-hover:via-border/30 group-hover:to-primary/20 transition-all duration-500">
                  <div className="bg-card/60 backdrop-blur-xl rounded-2xl overflow-hidden">
                    {/* Video Player */}
                    <div className="aspect-video">
                      {video.hlsUrl ? (
                        <HeroVideoPlayer
                          hlsUrl={video.hlsUrl}
                          thumbnailUrl={video.thumbnailUrl}
                          previewUrl={video.previewUrl}
                          className="w-full h-full"
                        />
                      ) : (
                        // Placeholder when no video URL
                        <div className="w-full h-full bg-gradient-to-br from-primary/10 via-background to-primary/5 flex items-center justify-center">
                          <div className="text-center">
                            <Play className="w-12 h-12 text-primary/50 mx-auto mb-2" />
                            <span className="text-sm text-muted-foreground">
                              Video coming soon
                            </span>
                          </div>
                        </div>
                      )}
                    </div>

                    {/* Video Info */}
                    <div className="p-5">
                      <h3 className="font-semibold text-lg mb-1">
                        {video.title}
                      </h3>
                      <p className="text-sm text-muted-foreground">
                        {video.description}
                      </p>
                    </div>
                  </div>
                </div>
              </motion.div>
            ))}
          </motion.div>
        </div>
      </section>
    );
  }
  ```

  Key features:
  - Reuses HeroVideoPlayer for click-to-play behavior
  - Graceful degradation if videos not configured
  - Responsive grid (1-col mobile, 2-col tablet, 3-col desktop)
  - Framer Motion animations matching existing Results page
  - Card styling consistent with testimonial cards
